Supabase Table Schemas

EMPLOYEES
create table public.employees (
  uuid uuid not null default gen_random_uuid (),
  internal_no integer null,
  employee_code text not null,
  employee_name text not null,
  department text null,
  start_date date null,
  basic numeric(14, 2) not null default 0,
  current_salary numeric(14, 2) null,
  skill_grade_id uuid null,
  seniority_grade_id uuid null,
  active boolean not null default true,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone not null default now(),
  preferred_name text null,
  fingerprint_id text null,
  probation boolean not null default false,
  position_id uuid not null,
  gets_bpjs_jp boolean not null default true,
  thr_preference text not null default 'balinese'::text,
  cash_loan_balance_idr numeric(14, 2) not null default 0,
  housing_allowance_idr numeric(14, 2) not null default 0,
  gets_meal_allowance boolean not null default true,
  bank text null,
  bank_account text null,
  bank_account_name text null,
  constraint employees_pkey primary key (uuid),
  constraint employees_bank_account_unique unique (bank_account),
  constraint employees_employee_id_key unique (employee_code),
  constraint employees_internal_id_key unique (internal_no),
  constraint employees_position_id_fkey foreign KEY (position_id) references positions (id),
  constraint employees_skill_grade_id_fkey foreign KEY (skill_grade_id) references skill_grades (id),
  constraint employees_seniority_grade_id_fkey foreign KEY (seniority_grade_id) references seniority_grades (id),
  constraint employees_thr_religion_check check (
    (
      thr_preference = any (
        array[
          'muslim'::text,
          'christian'::text,
          'balinese'::text
        ]
      )
    )
  ),
  constraint employees_bank_account_name_no_trailing_spaces check (
    (
      (bank_account_name is null)
      or (bank_account_name = btrim(bank_account_name))
    )
  ),
  constraint employee_code_format check ((employee_code ~ '^[A-Za-z0-9]{3,12}$'::text))
) TABLESPACE pg_default;

create unique INDEX IF not exists employees_internal_no_unique on public.employees using btree (internal_no) TABLESPACE pg_default;

create unique INDEX IF not exists employees_employee_code_unique on public.employees using btree (employee_code) TABLESPACE pg_default;

create unique INDEX IF not exists employees_fingerprint_id_unique on public.employees using btree (fingerprint_id) TABLESPACE pg_default
where
  (fingerprint_id is not null);

create trigger set_employees_updated_at BEFORE
update on employees for EACH row
execute FUNCTION set_updated_at ();


PAYROLL_ENTRIES
create table public.payroll_entries (
  id uuid not null default gen_random_uuid (),
  period_id uuid not null,
  employee_uuid uuid not null,
  full_days_worked integer not null default 0,
  on_time_days integer not null default 0,
  excused_full_days integer not null default 0,
  unexcused_full_days integer not null default 0,
  notes text null,
  salary_to_pay numeric(14, 2) null,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone not null default now(),
  excused_half_days integer not null default 0,
  unexcused_half_days integer not null default 0,
  late_minutes_count integer not null default 0,
  meal_allowance_idr numeric(14, 2) not null default 0,
  loan_repayment_idr numeric(14, 2) not null default 0,
  new_loan_idr numeric(14, 2) not null default 0,
  overtime_hours_1 numeric(8, 2) not null default 0,
  overtime_hours_2 numeric(8, 2) not null default 0,
  overtime_hours_3 numeric(8, 2) not null default 0,
  other_adjustment_idr numeric(14, 2) not null default 0,
  other_adjustment_note text null,
  constraint payroll_entries_pkey primary key (id),
  constraint payroll_entries_period_id_employee_uuid_key unique (period_id, employee_uuid),
  constraint payroll_entries_employee_uuid_fkey foreign KEY (employee_uuid) references employees (uuid) on delete RESTRICT,
  constraint payroll_entries_period_id_fkey foreign KEY (period_id) references payroll_periods (id) on delete CASCADE
) TABLESPACE pg_default;

create index IF not exists idx_payroll_entries_period on public.payroll_entries using btree (period_id) TABLESPACE pg_default;

create index IF not exists idx_payroll_entries_employee on public.payroll_entries using btree (employee_uuid) TABLESPACE pg_default;

create trigger set_entries_updated_at BEFORE
update on payroll_entries for EACH row
execute FUNCTION set_updated_at ();


PAYROLL_PERIODS
create table public.payroll_periods (
  id uuid not null default gen_random_uuid (),
  year integer not null,
  month integer not null,
  working_days integer not null,
  locked boolean not null default false,
  notes text null,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone not null default now(),
  red_days integer not null default 0,
  constraint payroll_periods_pkey primary key (id),
  constraint payroll_periods_year_month_key unique (year, month),
  constraint payroll_periods_month_check check (
    (
      (month >= 1)
      and (month <= 12)
    )
  ),
  constraint payroll_periods_planned_working_days_check check ((working_days >= 0)),
  constraint payroll_periods_red_days_check check (
    (
      (red_days >= 0)
      and (red_days <= 31)
    )
  ),
  constraint payroll_periods_red_days_range check (
    (
      (red_days >= 0)
      and (red_days <= 31)
    )
  )
) TABLESPACE pg_default;

create trigger set_periods_updated_at BEFORE
update on payroll_periods for EACH row
execute FUNCTION set_updated_at ();

  updated_at timestamp with time zone not null default now(),

  lateness_base_deduction_idr numeric(14, 2) not null default 25000,
  lateness_base_minutes integer not null default 5,
  lateness_increment_idr numeric(14, 2) not null default 10000,
  lateness_increment_minutes integer not null default 5,
  constraint payroll_settings_pkey primary key (id),
  constraint payroll_settings_hours_per_day_chk check (
    (
      (hours_per_day >= 1)
      and (hours_per_day <= 24)
    )
  ),
  constraint payroll_settings_payroll_end_date_day_chk check (
    (
      (payroll_end_date >= 1)
      and (payroll_end_date <= 31)
    )
  ),
  constraint payroll_settings_standard_working_days_chk check (
    (
      (standard_working_days >= 1)
      and (standard_working_days <= 31)
    )
  )
) TABLESPACE pg_default;


POSITIONS
create table public.positions (
  id uuid not null default gen_random_uuid (),
  name text not null,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone not null default now(),
  allowance_idr numeric(14, 2) not null default 0,
  constraint positions_pkey primary key (id),
  constraint positions_name_key unique (name)
) TABLESPACE pg_default;

create unique INDEX IF not exists positions_name_lower_unique on public.positions using btree (
  lower(
    TRIM(
      both
      from
        name
    )
  )
) TABLESPACE pg_default;


PROFILES
create table public.profiles (
  id uuid not null,
  role text not null default 'viewer'::text,
  created_at timestamp with time zone not null default now(),
  constraint profiles_pkey primary key (id),
  constraint profiles_id_fkey foreign KEY (id) references auth.users (id) on delete CASCADE,
  constraint profiles_role_check check (
    (
      role = any (
        array['admin'::text, 'payroll'::text, 'viewer'::text]
      )
    )
  )
) TABLESPACE pg_default;

SKILL_GRADES
create table public.skill_grades (
  id uuid not null default gen_random_uuid (),
  notes text null,
  created_at timestamp with time zone not null default now(),
  level numeric(4, 2) not null,
  increase_monthly_idr integer not null default 0,
  position_id uuid not null,
  constraint skill_grades_pkey primary key (id),
  constraint skill_grades_position_id_fkey foreign KEY (position_id) references positions (id),
  constraint skill_grades_level_positive check ((level > (0)::numeric))
) TABLESPACE pg_default;

create unique INDEX IF not exists skill_grades_position_id_level_unique on public.skill_grades using btree (position_id, level) TABLESPACE pg_default;

create trigger enforce_min_skill_grade BEFORE DELETE on skill_grades for EACH row
execute FUNCTION prevent_delete_last_skill_grade ();
